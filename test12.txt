1. Ansible — Install and Start Nginx
---
- name: Install and start Nginx
  hosts: all
  become: true

  tasks:
    - name: Install Nginx
      ansible.builtin.package:
        name: nginx
        state: present

    - name: Start and enable Nginx
      ansible.builtin.service:
        name: nginx
        state: started
        enabled: true

3. Kubernetes — Deployment with 3 Replicas
apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-deployment
spec:
  replicas: 3
  selector:
    matchLabels:
      app: nginx
  template:
    metadata:
      labels:
        app: nginx
    spec:
      containers:
        - name: nginx
          image: nginx
          ports:
            - containerPort: 80

5. Jenkins Pipeline — Deploy to Multiple Environments
pipeline {
    agent any

    environment {
        INSTITUTE_NAME = 'Pathnex'
    }

    stages {
        stage('Deploy to Dev') {
            steps {
                sh 'echo "Deploying $INSTITUTE_NAME to Dev environment"'
            }
        }

        stage('Deploy to Staging') {
            steps {
                input message: 'Approve Staging Deployment?'

                sh 'echo "Deploying $INSTITUTE_NAME to Staging environment"'
            }
        }

        stage('Deploy to Prod') {
            steps {
                input message: 'Approve Prod Deployment?'

                sh 'echo "Deploying $INSTITUTE_NAME to Prod environment"'
            }
        }
    }
}
